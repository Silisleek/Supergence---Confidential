export class TokenBucket {
  constructor(capacity, refillRatePerSec) {
    this.capacity = capacity;
    this.refillRate = refillRatePerSec;
    this.tokens = capacity;
    this.lastRefill = Date.now();
  }

  refill() {
    const now = Date.now();
    const elapsed = (now - this.lastRefill) / 1000;
    // BUG 1: Sub-millisecond rapid calls see elapsed === 0
    // causing token starvation on burst traffic!
    this.tokens = Math.min(this.capacity, this.tokens + elapsed * this.refillRate);
    this.lastRefill = now;
  }

  tryConsume(cost = 1) {
    this.refill();
    // BUG 2: Decrements BEFORE checking sufficiency!
    // Tokens become negative (-4) and leak!
    this.tokens -= cost;
    if (this.tokens < 0) {
      return false; 
    }
    return true;
  }
}
